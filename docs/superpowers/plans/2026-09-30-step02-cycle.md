## 关键决策（本计划据此编写）

| 决策 | 内容 | 理由 |
|---|---|---|
| 预测门槛 | 至少 2 个完整周期（即 3 次月经开始日）才给出日期；否则状态为 `insufficient_data`，不给任何日期 | 验收第 3 条要求数据不足时不给假精确 |
| 周期均值 | 取最近 6 个周期，按新旧线性加权（最旧权重 1，最新权重 n） | 规格要求近期权重更高 |
| 区间半宽 | `max(2, ceil(1.5 × 加权标准差))` 天；周期少于 4 个时再加 1 天 | 波动越大区间越宽；数据少时更保守 |
| 排卵日 | 下次月经开始日减 14 天（黄体期按 14 天估算）；易孕窗口为排卵日前 5 天到后 1 天 | 常用日历法口径，文档中写明误差大 |
| 降级标记 | `few_cycles`、`irregular`（最长最短相差超过 7 天）、`cycle_out_of_range`（任一周期短于 21 或长于 35 天）、`overdue`（今天晚于预测区间上限）、`long_gap`（距上次开始超过 90 天） | 规格要求异常数据打"可能不准"标记并建议咨询医生 |
| 阶段划分 | 以周期第几天计：经期天数内为月经期；排卵日前后各 1 天为排卵期；两者之间为卵泡期；排卵期之后为黄体期 | 简单、可手算核对 |
| 暂停 | 独立的经期设置表，状态为 `tracking`、`pregnant`、`paused`；非 `tracking` 时不预测、不判断阶段 | 规格第 8 条 |
| 提示规则 | 独立数据文件，每条含编号、适用阶段、类别（营养学、传统经验、生活方式）、严重程度、标题、内容、依据；传统经验单独归类 | 规格要求规则可维护、可逐条审核 |
| 日期存储 | 起止日期明文，每日记录内容整体加密 | 预测需要按日期查询；加密列无法排序和过滤 |

"""经期统计、预测与阶段判断。纯计算，不访问数据库，便于测试，也供后续的 SDK 直接复用。"""

import math
from collections import Counter
from dataclasses import dataclass, field
from datetime import date, timedelta
from typing import Any

LUTEAL_DAYS = 14  # 黄体期按 14 天估算：排卵日 = 下次月经开始日 - 14 天
MAX_CYCLES_USED = 6  # 只用最近 6 个周期
MIN_CYCLES_FOR_PREDICTION = 2  # 至少 2 个完整周期（3 次月经开始日）才给出预测日期
FEW_CYCLES_THRESHOLD = 4  # 少于 4 个周期时区间再加宽 1 天
MIN_HALF_WIDTH_DAYS = 2
NORMAL_CYCLE_MIN = 21
NORMAL_CYCLE_MAX = 35
IRREGULAR_SPREAD_DAYS = 7  # 最长与最短周期相差超过 7 天视为不规律
LONG_GAP_DAYS = 90
DEFAULT_PERIOD_LENGTH = 5
BLEEDING_FLOWS = frozenset({"spotting", "light", "medium", "heavy"})

FLAG_NOTES = {
    "few_cycles": "记录的周期还比较少，预测仅供参考，预测区间已相应放宽。",
    "irregular": "近几次周期长短差异较大，预测区间已相应放宽。",
    "cycle_out_of_range": "有周期短于 21 天或长于 35 天，预测可能不准；如果这种情况持续出现，建议咨询医生。",
    "overdue": "已经超过预计的月经开始时间；如果推迟持续或伴随不适，建议咨询医生。",
    "long_gap": "距离上次记录的月经已经超过 90 天，预测可能不准；如有需要请咨询医生。",
}
PREDICTION_DISCLAIMER = "预测仅供参考，不能用于避孕或诊断。"
INSUFFICIENT_DATA_NOTE = "至少需要记录 3 次月经的开始日期，才能给出预测。"

    logged_days: int
    top_symptoms: list[SymptomCount]
    average_pain_on_bleeding_days: float | None


def round_half_up(value: float) -> int:
    return math.floor(value + 0.5)


def _sorted(periods: list[Period]) -> list[Period]:
    return sorted(periods, key=lambda period: period.start)


def cycle_lengths(periods: list[Period]) -> list[int]:
    """相邻两次月经开始日之间的天数，按时间先后排列。"""
    ordered = _sorted(periods)
    return [(later.start - earlier.start).days for earlier, later in zip(ordered, ordered[1:])]


def weighted_mean_std(values: list[int]) -> tuple[float, float]:
    """线性加权：越近的周期权重越高（最旧为 1，最新为 n）。"""
    weights = range(1, len(values) + 1)
    total = sum(weights)
    mean = sum(w * v for w, v in zip(weights, values)) / total
    variance = sum(w * (v - mean) ** 2 for w, v in zip(weights, values)) / total
    return mean, math.sqrt(variance)


def compute_stats(periods: list[Period]) -> CycleStats:
    all_cycles = cycle_lengths(periods)
    recent = all_cycles[-MAX_CYCLES_USED:]
    lengths = [(p.end - p.start).days + 1 for p in _sorted(periods) if p.end is not None][-MAX_CYCLES_USED:]
    mean = std = None
    if recent:
        mean, std = weighted_mean_std(recent)
    return CycleStats(
        cycle_count=len(all_cycles),
        period_count=len(periods),
        avg_cycle_length=None if mean is None else round(mean, 1),
        cycle_length_std=None if std is None else round(std, 1),
        shortest_cycle=min(recent) if recent else None,
        longest_cycle=max(recent) if recent else None,
        avg_period_length=round(sum(lengths) / len(lengths), 1) if lengths else None,
        recent_cycle_lengths=recent,
    )


def predict(periods: list[Period], today: date) -> Prediction:
    recent = cycle_lengths(periods)[-MAX_CYCLES_USED:]
    if len(recent) < MIN_CYCLES_FOR_PREDICTION:
        return Prediction(status="insufficient_data", notes=[INSUFFICIENT_DATA_NOTE])

    mean, std = weighted_mean_std(recent)
    half_width = max(MIN_HALF_WIDTH_DAYS, math.ceil(1.5 * std))
    flags: list[str] = []
    if len(recent) < FEW_CYCLES_THRESHOLD:
        half_width += 1
        flags.append("few_cycles")
    if max(recent) - min(recent) > IRREGULAR_SPREAD_DAYS:
        flags.append("irregular")
    if any(n < NORMAL_CYCLE_MIN or n > NORMAL_CYCLE_MAX for n in recent):
        flags.append("cycle_out_of_range")

    last_start = _sorted(periods)[-1].start
    next_start = last_start + timedelta(days=round_half_up(mean))
    window = timedelta(days=half_width)
    if today > next_start + window:
        flags.append("overdue")
    if (today - last_start).days > LONG_GAP_DAYS:
        flags.append("long_gap")

    ovulation = next_start - timedelta(days=LUTEAL_DAYS)
    return Prediction(
        status="ok",
        next_period_start=next_start,
        next_period_range=DateRange(next_start - window, next_start + window),
        ovulation_day=ovulation,
        ovulation_range=DateRange(ovulation - window, ovulation + window),
        fertile_window=DateRange(ovulation - timedelta(days=5), ovulation + timedelta(days=1)),
        flags=flags,
        notes=[FLAG_NOTES[flag] for flag in flags],
    )


def summarize_symptoms(day_details: list[dict[str, Any]]) -> SymptomSummary:
    """统计每日记录里出现最多的症状（同一天重复只算一次），以及出血日的平均疼痛程度。"""
    counter = Counter(symptom for details in day_details for symptom in dict.fromkeys(details.get("symptoms") or []))
    pains = [
        details["pain"]
        for details in day_details
        if details.get("flow") in BLEEDING_FLOWS and details.get("pain") is not None
    ]
    return SymptomSummary(
        logged_days=len(day_details),
        top_symptoms=[SymptomCount(symptom, days) for symptom, days in counter.most_common(5)],
        average_pain_on_bleeding_days=round(sum(pains) / len(pains), 1) if pains else None,
    )

  - `app.cycle.engine.PhaseInfo(day: date, phase: str, cycle_day: int | None = None, days_until_next_period: int | None = None, is_predicted: bool = False, reason: str | None = None)`；phase 取值为四个阶段或 `unknown`；reason 取值 `no_data`、`before_first_record`、`insufficient_data`，暂停时由任务 4 填 `pregnant` 或 `paused`

阶段规则（`is_predicted` 的含义：这一天所在周期的下一次月经还没有记录，阶段是按预测推算的）：
1. 没有任何记录 → `unknown`，reason `no_data`；日期早于第一次记录 → `unknown`，reason `before_first_record`。
2. 锚点是开始日不晚于该日期的最近一次月经。
3. 锚点之后已有下一次记录：周期长度用真实值，`is_predicted` 为假。
4. 锚点是最近一次：周期不足 2 个时，只能判断是否在经期内（在则 `menstrual`，否则 `unknown`，reason `insufficient_data`，但给出 `cycle_day`）；周期足够时用加权均值四舍五入作为周期长度，超出一个周期长度的日期按周期长度向后推算，`is_predicted` 为真。
5. 经期天数：该周期有结束日就用真实天数；没有结束日或是推算出的周期，用平均经期天数四舍五入（没有则 5 天）。
6. 排卵日在周期中的第几天 = max(周期长度 − 14 + 1, 经期天数 + 2)；第 1 天到经期最后一天为 `menstrual`；排卵日前 1 天到后 1 天为 `ovulation`；两者之间为 `follicular`；之后为 `luteal`。
7. `days_until_next_period` = 本周期开始日 + 周期长度 − 该日期。

按规律型样例手算（最近一次开始 08-22，经期 08-22 至 08-26，周期长度 28，排卵日为第 15 天）：08-22 第 1 天月经期、距下次 28 天；08-26 第 5 天月经期；08-27 第 6 天卵泡期；09-03 第 13 天卵泡期；09-04 第 14 天排卵期；09-06 第 16 天排卵期；09-07 第 17 天黄体期；09-18 第 28 天黄体期、距下次 1 天；09-19 推算的新周期第 1 天月经期、距下次 28 天。已结束的周期 06-26（真实长度 29，排卵日为第 16 天）：07-10 第 15 天排卵期、非推算；07-24 第 29 天黄体期、距下次 1 天。

PHASES = ("menstrual", "follicular", "ovulation", "luteal")


@dataclass(frozen=True)
class PhaseInfo:
    day: date
    phase: str  # 四个阶段之一，或 unknown
    cycle_day: int | None = None
    days_until_next_period: int | None = None
    is_predicted: bool = False  # 所在周期的下一次月经尚未记录，阶段按预测推算
    reason: str | None = None  # phase 为 unknown 时说明原因


def _period_length(period: Period, typical: int) -> int:
    return (period.end - period.start).days + 1 if period.end is not None else typical


def get_phase(periods: list[Period], day: date) -> PhaseInfo:
    ordered = _sorted(periods)
    if not ordered:
        return PhaseInfo(day=day, phase="unknown", reason="no_data")
    past = [period for period in ordered if period.start <= day]
    if not past:
        return PhaseInfo(day=day, phase="unknown", reason="before_first_record")

    anchor = past[-1]
    later = ordered[len(past):]
    stats = compute_stats(ordered)
    typical_period = (
        round_half_up(stats.avg_period_length) if stats.avg_period_length else DEFAULT_PERIOD_LENGTH
    )

    if later:
        cycle_start = anchor.start
        cycle_length = (later[0].start - anchor.start).days
        menstrual_days = _period_length(anchor, typical_period)
        is_predicted = False
    else:
        recent = stats.recent_cycle_lengths
        if len(recent) < MIN_CYCLES_FOR_PREDICTION:
            cycle_day = (day - anchor.start).days + 1
            if cycle_day <= _period_length(anchor, typical_period):
                return PhaseInfo(day=day, phase="menstrual", cycle_day=cycle_day, is_predicted=anchor.end is None)
            return PhaseInfo(day=day, phase="unknown", cycle_day=cycle_day, reason="insufficient_data")
        cycle_length = round_half_up(weighted_mean_std(recent)[0])
        cycles_ahead = (day - anchor.start).days // cycle_length
        cycle_start = anchor.start + timedelta(days=cycles_ahead * cycle_length)
        menstrual_days = _period_length(anchor, typical_period) if cycles_ahead == 0 else typical_period
        is_predicted = True

    cycle_day = (day - cycle_start).days + 1
    ovulation_cycle_day = max(cycle_length - LUTEAL_DAYS + 1, menstrual_days + 2)
    if cycle_day <= menstrual_days:
        phase = "menstrual"
    elif cycle_day < ovulation_cycle_day - 1:
        phase = "follicular"
    elif cycle_day <= ovulation_cycle_day + 1:
        phase = "ovulation"
    else:
        phase = "luteal"
    return PhaseInfo(
        day=day,
        phase=phase,
        cycle_day=cycle_day,
        days_until_next_period=(cycle_start + timedelta(days=cycle_length) - day).days,
        is_predicted=is_predicted,
    )

```python
from functools import lru_cache
from pathlib import Path
from typing import Literal

from pydantic import BaseModel, ConfigDict, TypeAdapter

RULES_FILE = Path(__file__).with_name("tips_rules.json")
TIPS_DISCLAIMER = "以上内容是一般性的生活方式建议，不构成医疗建议，也不能用于诊断；身体不适请及时就医。"


class TipRule(BaseModel):
    model_config = ConfigDict(extra="forbid", frozen=True)

    id: str
    phase: Literal["menstrual", "follicular", "ovulation", "luteal", "any"]
    category: Literal["nutrition", "tradition", "lifestyle"]
    severity: Literal["info", "caution"]
    title: str
    content: str
    basis: str


@lru_cache
def load_tip_rules() -> tuple[TipRule, ...]:
    rules = TypeAdapter(list[TipRule]).validate_json(RULES_FILE.read_bytes())
    ids = [rule.id for rule in rules]
    if len(ids) != len(set(ids)):
        raise ValueError("tips_rules.json 中存在重复的规则编号")
    return tuple(rules)


def tips_for_phase(phase: str) -> list[TipRule]:
    """某阶段适用的提示，按规则文件中的顺序；阶段未知时只返回通用提示。"""
    return [rule for rule in load_tip_rules() if rule.phase in (phase, "any")]

经期记录校验规则（在 `records._validate_period` 一处实现）：
1. 开始日和结束日都不能晚于今天，否则 `period_in_future`。
2. 结束日不能早于开始日，否则 `invalid_period_range`。
3. 单次经期最多 15 天（结束日 − 开始日 ≤ 14），否则 `period_too_long`。
4. 与本人其他记录的日期范围不能重叠；没有结束日的记录视为一直持续到今天。冲突对象没有结束日时，提示"与一次尚未结束的经期重叠，请先填写那次经期的结束日期"，否则提示"与已有的经期记录日期重叠"。错误码均为 `period_overlap`。
（第 4 条同时保证了：最多只有一次没有结束日的记录，而且它一定是最近的一次。）

"""经期相关数据的读写与校验。所有查询都从 owned_by 开始。"""

import uuid
from datetime import date
from typing import Any

from sqlalchemy.dialects.postgresql import insert
from sqlalchemy.orm import Session

from app.common.ownership import get_owned_or_404, owned_by
from app.cycle.models import CyclePeriod, CycleSettings
from app.errors import AppError

MAX_PERIOD_DAYS = 15


def list_periods(db: Session, user_id: uuid.UUID) -> list[CyclePeriod]:
    return list(db.scalars(owned_by(CyclePeriod, user_id).order_by(CyclePeriod.start_date.desc())))


def _validate_period(
    db: Session,
    user_id: uuid.UUID,
    start: date,
    end: date | None,
    today: date,
    exclude_id: uuid.UUID | None = None,
) -> None:
    if start > today or (end is not None and end > today):
        raise AppError(422, "period_in_future", "经期日期不能晚于今天")
    if end is not None and end < start:
        raise AppError(422, "invalid_period_range", "结束日期不能早于开始日期")
    if end is not None and (end - start).days > MAX_PERIOD_DAYS - 1:
        raise AppError(422, "period_too_long", f"单次经期不能超过 {MAX_PERIOD_DAYS} 天，请检查日期")
    new_last = end or today  # 没有结束日视为一直持续到今天
    for other in list_periods(db, user_id):
        if other.id == exclude_id:
            continue
        other_last = other.end_date or today
        if start <= other_last and other.start_date <= new_last:
            if other.end_date is None:
                message = "与一次尚未结束的经期重叠，请先填写那次经期的结束日期"
            else:
                message = "与已有的经期记录日期重叠"
            raise AppError(409, "period_overlap", message)


def create_period(db: Session, user_id: uuid.UUID, start: date, end: date | None, today: date) -> CyclePeriod:
    _validate_period(db, user_id, start, end, today)
    period = CyclePeriod(user_id=user_id, start_date=start, end_date=end)
    db.add(period)
    db.commit()
    db.refresh(period)
    return period


def update_period(
    db: Session, user_id: uuid.UUID, period_id: uuid.UUID, changes: dict[str, Any], today: date
) -> CyclePeriod:
    period = get_owned_or_404(db, CyclePeriod, user_id, period_id)
    start = changes.get("start_date", period.start_date)
    end = changes["end_date"] if "end_date" in changes else period.end_date
    _validate_period(db, user_id, start, end, today, exclude_id=period.id)
    period.start_date, period.end_date = start, end
    db.commit()
    db.refresh(period)
    return period


def delete_period(db: Session, user_id: uuid.UUID, period_id: uuid.UUID) -> None:
    db.delete(get_owned_or_404(db, CyclePeriod, user_id, period_id))
    db.commit()


def get_cycle_settings(db: Session, user_id: uuid.UUID) -> CycleSettings:
    settings = db.scalar(owned_by(CycleSettings, user_id))
    if settings is None:
        # 与档案相同：并发的首次请求主键冲突时什么都不做，再查一次。
        db.execute(
            insert(CycleSettings)
            .values(user_id=user_id, status="tracking")
            .on_conflict_do_nothing(index_elements=[CycleSettings.user_id])
        )
        db.commit()
        settings = db.scalar(owned_by(CycleSettings, user_id))
    return settings


def set_cycle_status(db: Session, user_id: uuid.UUID, status: str) -> CycleSettings:
    settings = get_cycle_settings(db, user_id)
    settings.status = status
    db.commit()
    db.refresh(settings)
    return settings

def upsert_day_log(
    db: Session, user_id: uuid.UUID, log_date: date, details: dict[str, Any], today: date
) -> CycleDayLog:
    if log_date > today:
        raise AppError(422, "log_in_future", "不能记录未来日期")
    log = db.scalar(owned_by(CycleDayLog, user_id).where(CycleDayLog.log_date == log_date))
    if log is None:
        log = CycleDayLog(user_id=user_id, log_date=log_date, details=details)
        db.add(log)
    else:
        log.details = details  # 整条替换；加密列必须重新赋值才会被保存
    db.commit()
    db.refresh(log)
    return log


def list_day_logs(db: Session, user_id: uuid.UUID, start: date, end: date) -> list[CycleDayLog]:
    query = (
        owned_by(CycleDayLog, user_id)
        .where(CycleDayLog.log_date >= start, CycleDayLog.log_date <= end)
        .order_by(CycleDayLog.log_date)
    )
    return list(db.scalars(query))


def delete_day_log(db: Session, user_id: uuid.UUID, log_date: date) -> None:
    log = db.scalar(owned_by(CycleDayLog, user_id).where(CycleDayLog.log_date == log_date))
    if log is None:
        raise AppError(404, "not_found", "记录不存在")
    db.delete(log)
    db.commit()

"""把数据库里的经期数据交给引擎计算，并处理暂停状态。"""

import uuid
from dataclasses import asdict
from datetime import date, timedelta

from sqlalchemy.orm import Session

from app.cycle import engine, records
from app.cycle.schemas import CycleStatsOut, PhaseOut, TipOut, TipsOut
from app.cycle.tips import TIPS_DISCLAIMER, tips_for_phase

SYMPTOM_WINDOW_DAYS = 365
PAUSED_NOTES = {
    "pregnant": "已标记为孕期，经期预测已暂停。孕期相关问题请以产检医生的意见为准。",
    "paused": "经期预测已暂停，可以在经期设置中恢复。",
}


def engine_periods(db: Session, user_id: uuid.UUID) -> list[engine.Period]:
    return [engine.Period(p.start_date, p.end_date) for p in records.list_periods(db, user_id)]


def _paused_status(db: Session, user_id: uuid.UUID) -> str | None:
    status = records.get_cycle_settings(db, user_id).status
    return None if status == "tracking" else status


def get_stats(db: Session, user_id: uuid.UUID, today: date) -> CycleStatsOut:
    stats = engine.compute_stats(engine_periods(db, user_id))
    logs = records.list_day_logs(db, user_id, today - timedelta(days=SYMPTOM_WINDOW_DAYS), today)
    symptoms = engine.summarize_symptoms([log.details for log in logs])
    return CycleStatsOut(**asdict(stats), **asdict(symptoms))


def get_prediction(db: Session, user_id: uuid.UUID, today: date) -> engine.Prediction:
    paused = _paused_status(db, user_id)
    if paused:
        return engine.Prediction(status="paused", notes=[PAUSED_NOTES[paused]])
    return engine.predict(engine_periods(db, user_id), today)


def get_phase_info(db: Session, user_id: uuid.UUID, day: date) -> engine.PhaseInfo:
    paused = _paused_status(db, user_id)
    if paused:
        return engine.PhaseInfo(day=day, phase="unknown", reason=paused)
    return engine.get_phase(engine_periods(db, user_id), day)


def get_tips(db: Session, user_id: uuid.UUID, day: date) -> TipsOut:
    phase = get_phase_info(db, user_id, day)
    return TipsOut(
        phase=PhaseOut.model_validate(phase),
        tips=[TipOut.model_validate(rule) for rule in tips_for_phase(phase.phase)],
        disclaimer=TIPS_DISCLAIMER,
    )

"""演示数据：三位经期用户——规律型、波动型、数据不足型。用法见 README 的“经期模块演示”。"""

from datetime import date, timedelta

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.auth.models import User
from app.auth.service import register_user
from app.cycle import records
from app.db import get_sessionmaker
from app.disclaimer import service as disclaimer_service
from app.disclaimer.content import DISCLAIMER_VERSION
from app.profile.service import update_profile

DEMO_PASSWORD = "demo-pass-123"
DEMO_USERS = {
    "demo_regular": [28, 28, 29, 28],
    "demo_irregular": [24, 33, 27, 36, 31],
    "demo_sparse": [28],
}
LAST_START_DAYS_AGO = 10
PERIOD_DAYS = 5


def build_periods(cycle_lengths: list[int], today: date) -> list[tuple[date, date]]:
    """从最近一次（今天往前 10 天开始）倒推，每次经期 5 天。"""
    starts = [today - timedelta(days=LAST_START_DAYS_AGO)]
    for length in reversed(cycle_lengths):
        starts.insert(0, starts[0] - timedelta(days=length))
    return [(start, start + timedelta(days=PERIOD_DAYS - 1)) for start in starts]


def seed(db: Session, today: date) -> list[str]:
    created = []
    for username, lengths in DEMO_USERS.items():
        if db.scalar(select(User).where(User.username == username)) is not None:
            continue
        user = register_user(db, username, DEMO_PASSWORD)
        disclaimer_service.accept(db, user.id, DISCLAIMER_VERSION)
        update_profile(db, user.id, {"gender": "female"})
        for start, end in build_periods(lengths, today):
            records.create_period(db, user.id, start, end, today)
        created.append(username)
    return created


def main() -> None:
    db = get_sessionmaker()()
    try:
        created = seed(db, date.today())
    finally:
        db.close()
    if created:
        print(f"已创建演示账号：{'、'.join(created)}，密码均为 {DEMO_PASSWORD}")
    else:
        print("演示账号已存在，未重复创建。")


if __name__ == "__main__":
    main()
```
